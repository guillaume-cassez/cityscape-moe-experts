% Float placement (App. A uses \begin{table}[H]) and pretty rules — kept here
% because pandoc applies either YAML header-includes OR -H, not both.
\usepackage{float}
\floatplacement{figure}{H}
\usepackage{booktabs}
% Tableaux d'un seul bloc : \needspace{h} pousse le tabular entier à la page
% suivante quand le reste de page est trop court, au lieu de le laisser se faire
% scier par un longtable (tables coupées p5->p6, p6->p7, p9->p10, mesuré le 2026-10-03).
\usepackage{needspace}
\usepackage{array}   % colonnes >{...} du post-processeur de tableaux

% Robust rendering of the inline Unicode superscripts used in the Markdown
% (e.g. "6×10⁻⁵"): map them to LaTeX so the build does not depend on the main
% font carrying every glyph (XeLaTeX + Liberation Serif drops U+207B etc.).
% The math symbols actually used inline (± × − ≈ ≤ ≥ α β ε σ Δ →) ARE in
% Liberation Serif, so only the super/subscripts and two arrows need mapping.
\usepackage{newunicodechar}
\newunicodechar{⁻}{\textsuperscript{\textminus}}
\newunicodechar{⁰}{\textsuperscript{0}}
\newunicodechar{¹}{\textsuperscript{1}}
\newunicodechar{²}{\textsuperscript{2}}
\newunicodechar{³}{\textsuperscript{3}}
\newunicodechar{⁴}{\textsuperscript{4}}
\newunicodechar{⁵}{\textsuperscript{5}}
\newunicodechar{⁶}{\textsuperscript{6}}
\newunicodechar{⁷}{\textsuperscript{7}}
\newunicodechar{⁸}{\textsuperscript{8}}
\newunicodechar{⁹}{\textsuperscript{9}}
% Consensus veto notation "C⊘B" uses U+2298 (CIRCLED DIVISION SLASH), which
% Liberation Serif lacks — map it to the standard LaTeX math symbol \oslash.
\newunicodechar{⊘}{\ensuremath{\oslash}}
% Parity notation "naive↔precomputed" uses U+2194 (LEFT RIGHT ARROW), absent
% from Liberation Serif — map it to the math \leftrightarrow.
\newunicodechar{↔}{\ensuremath{\leftrightarrow}}

% --- Code-spans cassables (mesure 2026-10-01 sur les PDF publics) : un chemin
% monospace non cassable produit un overfull hbox qui SORT de la page (mots
% mesurés à x2 = 612-617 pt sur une page de 612 pt) : le lecteur perd la fin du
% chemin. On autorise la coupe après / . et _ dans tout \texttt (prose ET
% cellules de tableaux pandoc). Le \lowercase fabrique les caractères ACTIFS au
% moment du préambule : un \catcode changé plus tard ne retokenise pas ce bloc.
\makeatletter
\newcommand{\cs@brk}{\discretionary{}{}{}}
\begingroup\lccode`\~=`\/
\lowercase{\endgroup\def\cs@activesA{\catcode`\/=\active\def~{/\cs@brk}}}
\begingroup\lccode`\~=`\.
\lowercase{\endgroup\def\cs@activesB{\catcode`\.=\active\def~{.\cs@brk}}}
\newcommand{\cs@setup}{\cs@activesA\cs@activesB\def\_{\char95\cs@brk}}
\let\cs@texttt\texttt
\renewcommand{\texttt}{\begingroup\cs@setup\cs@texttt@i}
\newcommand{\cs@texttt@i}[1]{\cs@texttt{#1}\endgroup}
\makeatother

% --- Glyphes propres au paper3 (absents de Liberation Serif OU du lmmono des
% code-spans) — mappés vers des symboles LaTeX pour un build à 0 glyphe manquant.
\usepackage{amssymb}   % \checkmark
% U+2265 ≥ : présent dans Liberation Serif mais ABSENT de lmmono10-regular, la
% police des code-spans (`top_expert_share ≥ part_max`) — d'où le mapping global.
\newunicodechar{≥}{\ensuremath{\geq}}
% U+226A ≪ (« 0.00 ≪ −22.20 ») : absent de Liberation Serif.
\newunicodechar{≪}{\ensuremath{\ll}}
% U+2705 ✅ / U+274C ❌ : verdicts des tables T0-T4 (emoji, absents de la police).
\newunicodechar{✅}{\ensuremath{\checkmark}}
\newunicodechar{❌}{\ensuremath{\times}}
